第一篇 · DOM 操作全解

DOM（Document Object Model）是浏览器把 HTML 解析后生成的树形 JS 对象。操作 DOM 是前端工程师的基本功，也是性能优化的关键。

1. 元素选择全解

选择元素有两大类 API：传统 API和 Query API。Query API 更灵活，支持 CSS 选择器。

方法返回值是否实时兼容性推荐度
getElementById(id)单个 Element-全部⭐⭐⭐⭐⭐
getElementsByTagName(tag)HTMLCollection✅ 实时全部⭐⭐
getElementsByClassName(cls)HTMLCollection✅ 实时IE9+⭐⭐
querySelector(sel)单个 Element❌ 静态IE8+⭐⭐⭐⭐⭐
querySelectorAll(sel)NodeList❌ 静态IE8+⭐⭐⭐⭐⭐
实时 vs 静态：getElementsByXxx 返回的是实时集合（DOM 变化会自动更新），querySelectorAll 返回的是静态快照（DOM 变化不影响结果）。实时集合在循环中修改 DOM 可能导致死循环！ 
// ========== 传统 API ==========
// 通过 ID 获取（最快）
const app = document.getElementById('app');

// 通过标签名获取（返回实时 HTMLCollection）
const divs = document.getElementsByTagName('div');
divs[0];  // 第一个 div
divs.length; // 数量
// ⚠️ HTMLCollection 只能用数字索引和 length，没有 forEach/map 等方法

// 通过类名获取（返回实时 HTMLCollection）
const items = document.getElementsByClassName('item');

// ========== Query API（推荐！） ==========
// querySelector：返回第一个匹配的元素
document.querySelector('#app');           // ID 选择器
document.querySelector('.container');     // 类选择器
document.querySelector('div > .item');    // 子选择器
document.querySelector('input[type=text]'); // 属性选择器
document.querySelector('.a .b');          // 后代选择器
document.querySelector('.a, .b');         // 分组选择器
document.querySelector('.a + .b');        // 相邻兄弟
document.querySelector('.a ~ .b');        // 通用兄弟

// querySelectorAll：返回所有匹配的静态 NodeList
const allItems = document.querySelectorAll('.item');
// NodeList 有 forEach 方法（现代浏览器）
allItems.forEach(item => console.log(item));

// ========== 在元素范围内查询 ==========
const container = document.querySelector('.container');
const inner = container.querySelector('.item'); // 只在 container 内查找

// ========== 类数组转真数组 ==========
// NodeList / HTMLCollection 都不是真数组
const nodeList = document.querySelectorAll('div');

// 方法1：展开运算符（推荐）
[...nodeList].forEach(el => ...);

// 方法2：Array.from
Array.from(nodeList).forEach(el => ...);

// 方法3：Array.prototype.slice.call（老写法）
Array.prototype.slice.call(nodeList).forEach(el => ...);

const el = document.querySelector('.item');

// ========== 父子关系 ==========

3. 增删改操作

// ========== 创建元素 ==========
const div = document.createElement('div');
const p = document.createElement('p');
const text = document.createTextNode('文本内容');
const comment = document.createComment('注释');

// ========== 插入元素 ==========
const parent = document.querySelector('.parent');

// 末尾追加
parent.appendChild(div);

// 在参考元素之前插入
parent.insertBefore(div, referenceEl);

// 现代插入方法（ES2015+，更灵活）
referenceEl.before(div);            // 在 referenceEl 前面插入
referenceEl.after(div);             // 在 referenceEl 后面插入
referenceEl.append(div, p, text);   // 在 referenceEl 内部的末尾插入（可多个）
referenceEl.prepend(div);           // 在 referenceEl 内部的开头插入

// ========== 替换元素 ==========
parent.replaceChild(newEl, oldEl);
oldEl.replaceWith(newEl);  // 更简洁的写法

// ========== 删除元素 ==========
el.remove();                  // 直接删除自己（现代写法）
parent.removeChild(el);       // 通过父元素删除（传统写法）

// ========== 克隆元素 ==========
const clone = el.cloneNode(true);  // true = 深克隆（含子元素）
const shallow = el.cloneNode(false); // 浅克隆（只克隆自己）
// ⚠️ cloneNode 不会复制通过 addEventListener 添加的事件
// ⚠️ cloneNode 不会复制 el.dataset 的动态属性（HTML attribute 会复制）

// ========== innerHTML vs insertAdjacentHTML ==========
// innerHTML：替换元素内部所有内容
el.innerHTML = '<p>新内容</p>';

// insertAdjacentHTML：在指定位置插入 HTML（不破坏现有内容）
el.insertAdjacentHTML('beforebegin', '<p>前面</p>');  // 元素前面
el.insertAdjacentHTML('afterbegin', '<p>内部开头</p>'); // 内部开头
el.insertAdjacentHTML('beforeend', '<p>内部末尾</p>');   // 内部末尾
el.insertAdjacentHTML('afterend', '<p>后面</p>');      // 元素后面

// ========== DocumentFragment（批量插入优化） ==========
// 场景：循环插入 1000 个 li 到 ul
// ❌ 差的做法：每次循环都操作 DOM（触发 1000 次重排）
for (let i = 0; i < 1000; i++) {
  const li = document.createElement('li');
  li.textContent = `Item ${i}`;
  ul.appendChild(li); // 每次 appendChild 都触发重排！
}

// ✅ 好的做法：用 DocumentFragment 批量插入（只触发 1 次重排）
const fragment = document.createDocumentFragment();
for (let i = 0; i < 1000; i++) {
  const li = document.createElement('li');
  li.textContent = `Item ${i}`;
  fragment.appendChild(li); // 在内存中操作，不触发重排
}
ul.appendChild(fragment); // 一次性插入，只触发 1 次重排

5. DOM 性能优化

// ========== 原则1：减少 DOM 访问次数 ==========
// ❌ 差：每次循环都访问 DOM
for (let i = 0; i < 100; i++) {
  document.getElementById('list').innerHTML += '<li>' + i + '</li>';
}

// ✅ 好：缓存 DOM 引用
const list = document.getElementById('list');
let html = '';
for (let i = 0; i < 100; i++) {
  html += '<li>' + i + '</li>';
}
list.innerHTML = html;

// ========== 原则2：批量修改样式 ==========
// ❌ 差：多次修改触发多次重排
el.style.width = '100px';
el.style.height = '100px';
el.style.margin = '10px';

// ✅ 好：一次性修改
el.style.cssText = 'width: 100px; height: 100px; margin: 10px;';
// 或切换 class
el.classList.add('new-style');

// ========== 原则3：离线操作 ==========
// 将元素从 DOM 中取出，修改完再放回
const parent2 = el.parentNode;
const next = el.nextSibling;
parent2.removeChild(el);  // 离线
// ... 对 el 做大量修改 ...
parent2.insertBefore(el, next); // 放回

// ========== 原则4：使用 requestAnimationFrame ==========
// 动画用 requestAnimationFrame 而不是 setInterval
function animate() {
  // 修改样式
  el.style.transform = `translateX(${x}px)`;
  x += 1;
  if (x < 300) requestAnimationFrame(animate);
}

第二篇 · 事件机制全解

事件是前端与用户交互的基石。理解事件流、事件委托和事件对象，是写出高质量前端代码的前提。

1. 事件绑定的 3 种方式

❌ 方式1：内联事件（不要用）

<button onclick="handleClick()">点击</button>

缺点：HTML 与 JS 耦合、不支持多个监听器、作用域混乱

⚠️ 方式2：onXxx 属性（少用）

btn.onclick = handleClick;
btn.onclick = anotherFn; // 覆盖了上一个！

缺点：同一个事件只能绑一个处理器

✅ 方式3：addEventListener（推荐）

btn.addEventListener('click', fn);
btn.addEventListener('click', fn2);
// 两个都会执行！

优点：支持多个监听器、可配置、可移除

// ========== addEventListener 完整用法 ==========
btn.addEventListener('click', handler);                  // 基本
btn.addEventListener('click', handler, true);          // 捕获阶段触发（第3个参数）
btn.addEventListener('click', handler, {
  capture: true,    // 捕获阶段触发
  once: true,       // 只触发一次后自动移除
  passive: true      // 承诺不调用 preventDefault（滚动性能优化）
});

// ========== 移除事件监听 ==========
// 注意：函数必须是同一个引用！
function handleClick(e) {
  console.log('clicked');
}
btn.addEventListener('click', handleClick);
btn.removeEventListener('click', handleClick); // ✅ 移除成功

// ❌ 这样无法移除（匿名函数不是同一个引用）
btn.addEventListener('click', () => console.log('clicked'));
btn.removeEventListener('click', () => console.log('clicked')); // ❌ 移除失败！

2. 事件传播机制（重要！）

事件在 DOM 树中传播分三个阶段：捕获阶段 → 目标阶段 → 冒泡阶段。

// 事件传播示意图：
//
//     捕获阶段（Capture）       目标阶段       冒泡阶段（Bubble）
//         ↓                      ↓                 ↑
//  document → html → body → div#parent → div#child → div#parent → body → ...
//

// ========== 捕获 vs 冒泡 ==========
// 默认情况下，addEventListener 在冒泡阶段触发
parent.addEventListener('click', () => console.log('parent 冒泡'));
child.addEventListener('click', () => console.log('child 冒泡'));
// 点击 child 输出顺序：
// 1. 'child 冒泡'（目标阶段）
// 2. 'parent 冒泡'（冒泡阶段，child 的事件冒泡到 parent）

// 捕获阶段触发（第3个参数传 true）
parent.addEventListener('click', () => console.log('parent 捕获'), true);
child.addEventListener('click', () => console.log('child 冒泡'));
// 点击 child 输出顺序：
// 1. 'parent 捕获'（捕获阶段先到 parent）
// 2. 'child 冒泡'（目标阶段 + 冒泡）

// ========== 阻止传播 ==========
child.addEventListener('click', (e) => {
  e.stopPropagation();      // 阻止事件继续传播（捕获/冒泡都停）
  // e.stopImmediatePropagation();  // 不仅阻止传播，还阻止同一元素上的其他监听器
});

// ========== 阻止默认行为 ==========
// 阻止：表单提交跳转、a 标签跳转、右键菜单、复选框勾选等
link.addEventListener('click', (e) => {
  e.preventDefault();  // 阻止 a 标签跳转
});

form.addEventListener('submit', (e) => {
  e.preventDefault();  // 阻止表单提交（改用 AJAX）
});

3. 事件委托（核心技巧）

利用事件冒泡，将子元素的事件处理委托到父元素统一处理。优势：省内存（一个监听器替代 N 个）+ 动态元素自动绑定。

// ========== 基本事件委托 ==========
// 场景：ul 下有大量 li，点击每个 li 要高亮

// ❌ 差的做法：每个 li 都绑定（浪费内存，新增 li 还要重新绑定）
document.querySelectorAll('li').forEach(li => {
  li.addEventListener('click', handleClick);
});

// ✅ 好的做法：委托给 ul
const ul = document.querySelector('ul');
ul.addEventListener('click', (e) => {
  // e.target 是实际被点击的元素
  const li = e.target.closest('li'); // 找到最近的 li 祖先
  if (!li) return;                  // 点击的不是 li
  if (!ul.contains(li)) return;        // 不是当前 ul 的子元素

  // 处理点击
  li.classList.toggle('active');
});

// ========== 实战：动态列表的增删 ==========
// 新增 li 不需要重新绑定事件，事件委托天然支持！
function addItem(text) {
  const li = document.createElement('li');
  li.innerHTML = `${text} <button class="del">删除</button>`;
  ul.appendChild(li);
  // 不需要给 li 或 .del 绑定事件！委托已经覆盖了
}

// 区分点击的是 li 还是 删除按钮
ul.addEventListener('click', (e) => {
  if (e.target.matches('.del')) {
    // 点击了删除按钮
    e.target.closest('li').remove();
  } else {
    // 点击了 li
    const li = e.target.closest('li');
    li.classList.add('selected');
  }
});
closest 方法：从元素自身开始，沿着 DOM 树向上查找最近的匹配选择器的祖先（包括自己）。在事件委托中是定位目标元素的利器。

4. 事件对象详解

element.addEventListener('click', function(e) {
  // ========== 核心属性 ==========
  e.type;              'click'
  e.target;            // 实际触发事件的元素（最内层）
  e.currentTarget;     // 绑定事件的元素（this === currentTarget）
  e.eventPhase;        // 1=捕获, 2=目标, 3=冒泡
  e.timeStamp;         // 事件发生的时间戳
  e.bubbles;           // 事件是否冒泡（boolean）
  e.cancelable;        // 是否可以阻止默认行为（boolean）
  e.defaultPrevented;  // 是否已调用了 preventDefault（boolean）

  // ========== 核心方法 ==========
  e.preventDefault();           // 阻止默认行为
  e.stopPropagation();          // 阻止事件继续传播
  e.stopImmediatePropagation(); // 阻止传播 + 阻止同元素其他监听器

  // ========== 鼠标事件专用 ==========
  // MouseEvent
  e.clientX;   // 相对于浏览器视口的 X（不含滚动）
  e.clientY;   // 相对于浏览器视口的 Y
  e.pageX;     // 相对于整个文档的 X（含滚动偏移）
  e.pageY;     // 相对于整个文档的 Y
  e.screenX;   // 相对于屏幕的 X
  e.offsetX;   // 相对于目标元素左上角的 X
  e.button;    // 0=左键, 1=中键, 2=右键
  e.buttons;   // 位掩码：1=左键 2=右键 4=中键
  e.ctrlKey;   // 是否按着 Ctrl
  e.shiftKey;  // 是否按着 Shift
  e.altKey;    // 是否按着 Alt
  e.metaKey;   // 是否按着 Cmd/Win

  // 滚动事件
  e.deltaX;    // 水平滚动量（正=向右）
  e.deltaY;    // 垂直滚动量（正=向下）

  // ========== 键盘事件专用 ==========
  // KeyboardEvent
  e.key;       // 按下的键（'Enter', 'a', 'ArrowUp' 等）
  e.code;      // 物理键（'KeyA', 'Enter' 等）
  e.keyCode;   // 数字码（已废弃，别用了）
  e.repeat;    // 是否长按重复（boolean）

  // key 和 code 的区别：
//  - key：按的是什么字符。Shift+a → 'A'
//  - code：按的是哪个物理键。Shift+a → 'KeyA'
//  - 做快捷键推荐用 code（不受输入法/大小写影响）
//  - 做文本输入推荐用 key（要获取实际字符）
});

// ========== 左滑删除（移动端常见交互） ==========
let startX = 0;

  startX = e.touches[0].clientX;
});

el.addEventListener('touchmove', (e) => {
  const currentX = e.touches[0].clientX;
  const deltaX = currentX - startX;

  // 左滑（deltaX 为负）
  if (deltaX < 0) {

  }
});

el.addEventListener('touchend', (e) => {
  const endX = e.changedTouches[0].clientX;
  const delta = endX - startX;

  if (delta < -100) {
    // 左滑超过阈值，触发删除
    el.style.transform = 'translateX(-80px)';
  } else {
    // 回弹
    el.style.transform = 'translateX(0)';
  }
});

第三篇 · 箭头函数全解

箭头函数（Arrow Function）是 ES6 引入的简洁函数语法，它不仅让代码更短，更重要的是不绑定自己的 this（这是与普通函数最本质的区别）。

1. 语法全解

// ========== 完整形式 ==========
const add = (a, b) => {
  return a + b;
};

// ========== 变体1：单个参数可省略括号 ==========
const double = n => n * 2;
// 等价于：const double = (n) => n * 2;
// ⚠️ 无参数或多余1个参数必须写括号
const getRandom = () => Math.random();
const sum = (a, b) => a + b;

// ========== 变体2：省略花括号（隐式返回） ==========
// 省略花括号时，单行表达式自动作为返回值
const double2 = n => n * 2;         // 返回 n*2
const getUser = () => ({ name: '张三' }); // 返回对象必须包括号！
// const getUser = () => { name: '张三' };  // ❌ 这是代码块，不是对象！

// ========== 变体3：返回对象字面量 ==========
// ✅ 正确：用括号包裹
const makeUser = name => ({ name, age: 25 });
// ❌ 错误：花括号被解析为函数体
// const makeUser = name => { name, age: 25 }; // SyntaxError

// ========== 变体4：多行必须用花括号+显式 return ==========
const calculate = (a, b) => {
  const sum = a + b;
  const product = a * b;
  return { sum, product };
};

// ========== 变体5：立即执行箭头函数 ==========
const result = (() => {
  const data = heavyCalculation();
  return data * 2;
})();

2. this 继承机制（核心重点）

最重要的规则：箭头函数没有自己的 this，它的 this 由外层作用域决定（词法作用域）。一旦绑定，无法通过 call/apply/bind 改变。 
// ========== 对比演示 ==========
const obj = {
  name: 'Vue',

  // 普通函数：this 由调用方式决定
  regularFn: function() {
    console.log(this.name); // 'Vue'（obj 调用的）
  },

  // 箭头函数：this 继承外层（即 obj 所在作用域 → 这里是模块/全局）
  arrowFn: () => {
    console.log(this.name); // undefined（外层 this 是 undefined 或 window）
  }

// 场景2：setTimeout 中的 this ==========
const obj2 = {
  name: 'React',

  // ❌ 普通函数：setTimeout 里的 this 是 window/undefined
  delayLog1: function() {
    setTimeout(function() {
      console.log(this.name); // undefined（严格模式）
    }, 100);
  },

  // ✅ 箭头函数：继承 delayLog2 的 this（即 obj2）
  delayLog2: function() {
    setTimeout(() => {
      console.log(this.name); // 'React' ✅
    }, 100);
  },

  // ✅✅ 更优雅：外层也用箭头函数（如果外层也是箭头函数，this 继续往上找）
  delayLog3: function() {
    setTimeout(() => {
      console.log(this.name); // 'React' ✅
    }, 100);
  }
};

// ========== 箭头函数中 call/apply/bind 无效 ==========
const arrow = () => this.name;
arrow.call({ name: 'A' });  // 不生效！
arrow.apply({ name: 'B' }); // 不生效！
arrow.bind({ name: 'C' })(); // 不生效！

第四篇 · this 指向全解

this 是 JavaScript 中最容易混淆的概念之一。它的指向在调用时确定（动态绑定），而非定义时。掌握 this 的 4 种绑定规则，就能解决所有 this 问题。

1. 四种绑定规则（按优先级从高到低）

判断 this 指向的口诀：先看是不是 new 调用 → 再看 call/apply/bind → 看谁调用的 → 最后看箭头函数 → 默认绑定

规则1：new 绑定（最高优先级）

function Person(name) {
  this.name = name;
  console.log(this); // Person { name: '张三' }（全新的对象）
}

const p = new Person('张三');
// new 做了什么：
// 1. 创建全新空对象 {}
// 2. 这个对象的 __proto__ = Person.prototype
// 3. this = 这个新对象
// 4. 执行函数体（给 this 添加属性）
// 5. return this（如果函数没有返回对象）

规则2：显式绑定（call / apply / bind）

function greet() {
  return `Hello, ${this.name}`;
}

const obj1 = { name: '张三' };
const obj2 = { name: '李四' };

// call：参数一个个传
greet.call(obj1); // 'Hello, 张三'
greet.call(obj2); // 'Hello, 李四'

// apply：参数用数组传
greet.apply(obj1); // 'Hello, 张三'

// bind：返回一个新函数（this 永久绑定）
const greetZhang = greet.bind(obj1);
greetZhang();         // 'Hello, 张三'（永远绑定 obj1）
greetZhang.call(obj2); // 'Hello, 张三'（bind 的优先级高于 call！）

规则3：隐式绑定（谁调用指向谁）

const obj = {
  name: 'Vue',
  sayName() {
    console.log(this.name);
  }
};

obj.sayName(); // 'Vue'（this = obj，因为 obj 调用的）

// ⚠️ 隐式丢失
const fn = obj.sayName; // 把函数赋给变量
fn(); // undefined（this = window/undefined，不再是 obj！）

// ⚠️ 回调中的丢失
setTimeout(obj.sayName, 100); // undefined（setTimeout 内部调用，this 是 window）

规则4：默认绑定（兜底）

// 严格模式
function show() {
  'use strict';
  console.log(this); // undefined
}
show();

// 非严格模式
function show2() {
  console.log(this); // window（浏览器）
}
show2();

2. 箭头函数：唯一的例外

// 箭头函数不参与上述 4 条规则！
// 它的 this 在定义时确定（词法作用域），不可改变

const obj = {
  name: 'Test',
  // 箭头函数定义在对象字面量里
  arrow: () => this.name, // this 不是 obj！是外层作用域的 this
  // 普通函数
  normal() { return this.name; } // this = obj ✅
};

  name: 'A',

  },
  fn2() {

      console.log(this.name); // 'A' ✅
    }, 100);
  },
  fn3() {
    // 老写法：保存 this
    const self = this;
    setTimeout(function() {
      console.log(self.name); // 'A' ✅
    }, 100);
  }
};

第五篇 · 异步编程全解

JavaScript 是单线程语言，异步编程是其核心能力。从回调函数到 Promise 再到 async/await，JS 的异步方案越来越优雅。

1. Promise 深度全解

// ========== Promise 的三种状态 ==========
// pending（进行中）→ fulfilled（已成功）
// pending（进行中）→ rejected（已失败）
// 状态一旦改变就不可逆！

// ========== 创建 Promise ==========
const p = new Promise((resolve, reject) => {
  // resolve 和 reject 只能生效一次（第一个被调用的）
  setTimeout(() => {
    if (Math.random() > 0.5) {
      resolve('成功数据');
    } else {
      reject(new Error('失败原因'));
    }
  }, 1000);
});

// ========== 消费 Promise ==========
p.then(
  value => console.log('成功：', value),
  error => console.log('失败：', error.message)
);

// ========== Promise 链式调用 ==========
// 关键：then 里 return 的值会作为下一个 then 的参数
fetch('/api/user')
  .then(res => res.json())              // return Promise
  .then(user => fetch(`/api/orders?userId=${user.id}`))  // return Promise
  .then(res => res.json())              // return Promise
  .then(orders => console.log(orders))     // return 普通值
  .catch(err => console.error(err));     // 捕获链中任何错误

// ========== Promise 的静态方法 ==========
// Promise.resolve：创建一个已成功的 Promise
Promise.resolve('直接成功').then(v => console.log(v));

// Promise.reject：创建一个已失败的 Promise
Promise.reject(new Error('直接失败')).catch(e => console.log(e));

// Promise.all：全部成功才成功，一个失败就失败
Promise.all([p1, p2, p3])
  .then(([r1, r2, r3]) => { })
  .catch(err => { }); // 任何一个失败就进 catch

// Promise.race：谁先完成用谁（不论成败）
Promise.race([fetchData(), timeout(5000)])
  .then(result => { });

// Promise.allSettled：等全部完成（不论成败）
Promise.allSettled([p1, p2, p3])
  .then(results => {
    results.forEach(r => {
      if (r.status === 'fulfilled') {
        console.log('成功：', r.value);
      } else {
        console.log('失败：', r.reason);
      }
    });
  });

// Promise.any：任何一个成功就成功（ES2021）
Promise.any([p1, p2, p3])
  .then(firstSuccess => { })  // 第一个成功的
  .catch(err => { });            // 全部失败才进 catch

  const data = await res.json();
  return data; // 等价于 return Promise.resolve(data)
}

async function load() {

    return data;
  } catch (err) {

    return defaultValue; // 返回默认值

  }

// ========== await 的并发优化 ==========
// ❌ 串行（慢）：总耗时 = t1 + t2 + t3
async function slow() {
  const a = await fetchA(); // 等 1 秒
  const b = await fetchB(); // 等 1 秒
  const c = await fetchC(); // 等 1 秒
  // 总耗时 3 秒
}

// ✅ 并发（快）：总耗时 = max(t1, t2, t3)
async function fast() {
  const [a, b, c] = await Promise.all([
    fetchA(), fetchB(), fetchC()
  ]);
  // 总耗时 1 秒
}

// ✅ 先启动再 await（另一种并发写法）
async function fast2() {
  const pA = fetchA(); // 立即发起，不等结果
  const pB = fetchB(); // 立即发起，不等结果
  const pC = fetchC(); // 立即发起，不等结果
  const a = await pA;  // 等结果
  const b = await pB;
  const c = await pC;
}

// ========== await 在循环中的行为 ==========
// for 循环中的 await 是串行的（一个完成再下一个）
async function sequential(urls) {
  for (const url of urls) {
    const data = await fetch(url); // 串行
  }
}

// 并发循环
async function parallel(urls) {
  const promises = urls.map(url => fetch(url));
  const results = await Promise.all(promises);
}

// ========== top-level await（ES2022） ==========
// 在模块顶层直接使用 await（不需要 async 函数包裹）

// Generator 是可以暂停和恢复的函数 ==========

// ========== 模式1：超时控制 ==========
function withTimeout(promise, ms = 5000) {
  const timeout = new Promise((_, reject) =>
    setTimeout(() => reject(new Error(`请求超时（${ms}ms）`)), ms)
  );
  return Promise.race([promise, timeout]);
}

// ========== 模式2：自动重试（指数退避） ==========
async function retry(fn, maxRetries = 3, baseDelay = 1000) {
  for (let i = 0; i < maxRetries; i++) {
    try {
      return await fn();
    } catch (err) {
      if (i === maxRetries - 1) throw err;
      const delay = baseDelay * 2 ** i; // 指数退避：1s, 2s, 4s
      await new Promise(r => setTimeout(r, delay));
    }
  }
}

// ========== 模式3：并发限制（控制同时进行的请求数） ==========
async function pool(tasks, limit) {
  const results = [];
  const executing = new Set();
  for (const task of tasks) {
    const p = task().then(r => { executing.delete(p); return r; });
    results.push(p);
    executing.add(p);
    if (executing.size >= limit) await Promise.race(executing);
  }
  return await Promise.all(results);
}

3. Proxy 与 Reflect

// ========== Proxy：拦截对象操作 ==========
const target = { name: '张三', age: 25 };

const proxy = new Proxy(target, {
  // get：拦截读取
  get(obj, prop, receiver) {
    if (!(prop in obj)) {
      throw new Error(`属性 ${prop} 不存在`);
    }
    return Reflect.get(obj, prop, receiver);
  },

  // set：拦截写入
  set(obj, prop, value, receiver) {
    if (prop === 'age') {
      if (typeof value !== 'number' || value < 0) {
        throw new Error('年龄必须是正数');
      }
    }
    return Reflect.set(obj, prop, value, receiver);
  },

  // has：拦截 in 运算符
  has(obj, prop) {
    if (prop.startsWith('_')) return false; // 隐藏私有属性
    return prop in obj;
  },

  // deleteProperty：拦截 delete
  deleteProperty(obj, prop) {
    if (prop === 'id') {
      throw new Error('id 不允许删除');
    }
    return delete obj[prop];
  }
});

// ========== Proxy 实际用途 ==========
// 用途1：数据校验
// 用途2：数据监听（Vue 3 响应式原理）
// 用途3：日志记录
// 用途4：负索引数组
function createNegativeArray(...elements) {
  return new Proxy(elements, {
    get(target, prop) {
      const index = Number(prop);
      if (index < 0) prop = target.length + index;
      return target[prop];
    }
  });
}
