第1章 · JavaScript 是什么？

JavaScript（简称 JS）是浏览器的编程语言，让网页"动起来"。它是前端三剑客之一：

语言职责类比
HTML内容结构骨架
CSS外观样式皮肤
JavaScript交互行为肌肉

JS 和 Java 的关系

没有关系。 就像雷锋和雷峰塔的关系。名字相似是历史原因（1995 年 Netscape 为了蹭 Java 热度取的）。

维度JavaJavaScript
类型系统静态强类型动态弱类型
运行环境JVM浏览器 / Node.js
面向对象基于类（class）基于原型（prototype）+ class 语法糖
编译编译为字节码解释执行（JIT 优化）
运行方式启动后运行整个程序逐行执行，永不报编译错误
变量声明时需指定类型赋值时自动推断类型
前端开发中的 JS：操作 DOM（页面元素）、处理用户交互（点击/输入/滚动）、发送网络请求（向后端 API 获取数据）、数据校验、动画效果、存储数据（localStorage）……几乎所有"能动"的东西都靠 JS。

在哪里写 JS 代码

/* 方式1：内嵌在 HTML 的 <script> 标签中 */
<script>
  console.log('Hello, JavaScript!');
</script>

/* 方式2：外部 JS 文件（推荐！） */
<script src="app.js"></script>

/* 方式3：浏览器控制台（F12 → Console），适合快速验证 */
console.log('快速测试');  // 直接按 Enter 执行
注意：CSS 写在 <style> 里，JS 写在 <script> 里。Vue 项目中，JS 写在 <script setup> 里。

实战：深拷贝的几种方案

// 方式1：JSON 序列化（最常用，但有局限）
const original = {
  name: '张三',
  age: 25,
  hobbies: ['读书', '写代码'],
  address: { city: '北京' }
};

const copy = JSON.parse(JSON.stringify(original));
copy.address.city = '上海';
console.log(original.address.city);  // '北京'（深拷贝成功）

// ⚠️ JSON 深拷贝的局限性：
const withFunc = {
  name: 'test',
  fn: function() return 1;,
  date: new Date(),
  undef: undefined,
  nan: NaN,
  inf: Infinity,
  regex: /abc/
};

const bad = JSON.parse(JSON.stringify(withFunc));
// fn 没了 → undefined
// date 变成字符串 → "2024-01-01T00:00:00.000Z"
// undef 没了 → JSON 中 undefined 被忽略
// nan → null, inf → null（NaN/Infinity 被转成 null）
// regex → {}（正则变成空对象）

// 方式2：structuredClone（现代浏览器，Node 17+）
const good2 = structuredClone(withFunc);
// 支持 Date/RegExp/Map/Set/ArrayBuffer
// 仍然不支持函数

// 方式3：手写递归深拷贝（面试题 + 老项目）
function deepClone(obj, map = new WeakMap()) {
  // 非对象或 null，直接返回
  if (obj === null || typeof obj !== 'object') {
    return obj;
  }

  // 防止循环引用
  if (map.has(obj)) {
    return map.get(obj);
  }

  // 日期
  if (obj instanceof Date) return new Date(obj.getTime());

  // 正则
  if (obj instanceof RegExp) return new RegExp(obj);

  // 数组或对象
  const clone = Array.isArray(obj) ? [] : {};
  map.set(obj, clone);

  for (const key in obj) {
    if (obj.hasOwnProperty(key)) {
      clone[key] = deepClone(obj[key], map);
    }
  }

  return clone;
}

逻辑运算符与短路求值

// && 与：两边都为 true 才返回 true
true && true;     // true
true && false;    // false

// || 或：一边为 true 就返回 true
true || false;    // true
false || false;   // false

// ! 非：取反
!true;              // false

// && 短路：左边为 falsy 直接返回左边
const result = user && user.name;

// || 短路：左边为 falsy 返回右边（设置默认值）
const displayName = userName || '匿名';

// ?? 空值合并（ES2020）：只有 null/undefined 才用默认值
const count = data ?? 0;
0 || 100;  // 100（0 被当成了 falsy）
0 ?? 100;  // 0（0 不是 null/undefined）

可选链 ?.（ES2020）

const response = { data: { user: { name: '张三' } } };

// 以前的安全写法：
const old = response && response.data && response.data.user && response.data.user.name;

// 可选链写法：
const name = response?.data?.user?.name;  // '张三'
const bad = response?.data?.xxx?.yyy;      // undefined（不报错）

// 配合 ?? 提供默认值
const city = response?.data?.user?.city ?? '未知';

防抖（Debounce）与节流（Throttle）— 面试必考

// ========== 防抖：触发后等待 N 秒才执行，期间再触发重新计时 ==========
// 场景：搜索框输入、窗口 resize
function debounce(fn, delay = 300) {
  let timer = null;

  return function(...args) {
    // 清除之前的定时器
    clearTimeout(timer);

    // 重新设置定时器
    timer = setTimeout(() => {
      fn.apply(this, args);
    }, delay);
  };
}

// 使用：搜索框输入
const searchInput = document.querySelector('#search');
searchInput.addEventListener('input', debounce(function(e) {
  console.log('搜索：', e.target.value);
  // 发送搜索请求...
}, 500));

// ========== 节流：N 秒内只执行一次 ==========
// 场景：滚动事件、按钮点击
function throttle(fn, delay = 300) {
  let lastTime = 0;

  return function(...args) {
    const now = Date.now();
    if (now - lastTime >= delay) {
      lastTime = now;
      fn.apply(this, args);
    }
  };
}

// 使用：滚动加载
window.addEventListener('scroll', throttle(() => {
  const { scrollTop, scrollHeight, clientHeight } = document.documentElement;
  if (scrollTop + clientHeight >= scrollHeight - 100) {
    console.log('加载更多...');
  }
}, 200));

// ========== 区别总结 ==========
// 防抖："等你停了再做" → 搜索框、resize
// 节流："做了就等一下" → 滚动、按钮防重复点击

function findInTree(tree, id) {
  for (const node of tree) {
    if (node.id === id) return node;
    if (node.children) {
      const found = findInTree(node.children, id);
      if (found) return found;
    }
  }
  return null;
}

function flattenTree(tree) {
  return tree.reduce((acc, node) => {
    acc.push({ id: node.id, name: node.name });
    if (node.children) {
      acc.push(...flattenTree(node.children));
    }
    return acc;
  }, []);
}

// ========== 2. 自动重试 ==========
async function retry(fn, maxRetries = 3, delay = 1000) {
  for (let i = 0; i < maxRetries; i++) {
    try {
      return await fn();
    } catch (err) {
      if (i === maxRetries - 1) throw err; // 最后一次还失败就抛出
      console.log(`第 ${i + 1} 次失败，${delay }ms 后重试`);
      await new Promise(r => setTimeout(r, delay));
    }
  }
}

// 使用
const result = await retry(() => fetch('/api/data').then(r => r.json()), 3, 1000);

// ========== 3. 并发限制（限制同时进行的请求数） ==========
async function pool(tasks, limit) {
  const results = [];
  const executing = new Set();

  for (const task of tasks) {
    const p = task().then(result => {
      executing.delete(p);
      return result;
    });
    results.push(p);
    executing.add(p);

    if (executing.size >= limit) {
      await Promise.race(executing);
    }
  }

  return await Promise.all(results);
}

  urls.map(url => () => fetch(url).then(r => r.json())),
  5 // 并发上限为 5

    if (prop === 'age' && value < 0) {
      throw new Error('年龄不能为负数');
    }
    obj[prop] = value;
    return true;  // 表示设置成功
  }
});

proxy.name;       // '张三'
proxy.age = 30;    // 成功
// proxy.age = -1;  // Error: 年龄不能为负数
// proxy.unknown;   // Error: 属性 unknown 不存在
实际用途：Vue 3 的响应式系统就是用 Proxy 实现的！当数据变化时触发更新，不需要像 Vue 2 一样递归遍历所有属性。

第14章 · Class 与面向对象

// ES6 Class 语法（本质是原型的语法糖）
class Animal {
  // 构造函数
  constructor(name) {
    this.name = name;
  }

  // 方法自动加到 prototype 上
  speak() {
    return `${this.name} 发出声音`;
  }

  // 静态方法（只能用 ClassName.method() 调用）
  static info() {
    return '这是 Animal 类';
  }
}

// 实例化
const dog = new Animal('旺财');
dog.speak();     // '旺财 发出声音'
Animal.info();   // '这是 Animal 类'

// 继承
class Cat extends Animal {
  constructor(name, color) {
    super(name);    // 调用父类构造函数
    this.color = color;
  }

  speak() {         // 重写父类方法
    return `${this.name} 是 ${this.color} 色的，喵喵喵`;
  }

  // getter/setter（访问器）
  get description() {
    return `${this.color} 猫 ${this.name}`;
  }
}

const mimi = new Cat('咪咪', '橘');
mimi.speak();         // '咪咪 是 橘 色的，喵喵喵'
mimi.description;     // '橘 猫 咪咪'
与 Java 的差异：JS 类的底层仍然是 prototype 链，不像 Java 有真正的类。JS 的继承是"对象委托给对象"而不是"类继承类"。Vue 项目中，类用得不多（函数式 + setup 更常用），但了解 class 有助于看第三方库。
